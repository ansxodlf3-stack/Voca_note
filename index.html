function findBlueWords(ctx,ocrWords){ const c=ctx.canvas; const W=c.width,H=c.height; const data=ctx.getImageData(0,0,W,H).data; function blueAt(x,y){ x=Math.round(x); y=Math.round(y); if(x<0||y<0||x>=W||y>=H)return false; const i=(y*W+x)*4; const r=data[i],g=data[i+1],b=data[i+2]; return b>65 && b>r*1.18 && b>g*1.02 && b-r>28; } function nearbyBlue(x,y){ for(let dy=-2;dy<=2;dy++){ for(let dx=-2;dx<=2;dx++){ if(blueAt(x+dx,y+dy))return true; } } return false; } let result=[]; for(const word of ocrWords){ const b=word.bbox; const x=(b.x0+b.x1)/2; const y=(b.y0+b.y1)/2; const ww=b.x1-b.x0; const wh=b.y1-b.y0; if(ww<8||wh<5)continue; let best=0; // 글자 주변에 다양한 크기의 타원형 표시가 있는지 확인 for(const fx of [0.65,0.85,1.05,1.25]){ for(const fy of [1.0,1.3,1.6,1.9]){ const rx=ww*fx; const ry=wh*fy; let sectors=Array(8).fill(0); let total=0; for(let i=0;i<120;i++){ const a=2*Math.PI*i/120; const px=x+rx*Math.cos(a); const py=y+ry*Math.sin(a); if(nearbyBlue(px,py)){ const s=Math.floor(i/15); sectors[s]++; total++; } } const occupied=sectors.filter(v=>v>0).length; // 위쪽과 아래쪽 표시도 확인하여 밑줄 오인식을 줄임 const top=sectors[0]+sectors[1]+sectors[7]>0; const bottom=sectors[3]+sectors[4]+sectors[5]>0; if(occupied>=3 && top && bottom){ best=Math.max(best,occupied/8); } } } if(best>0){ result.push(word.text.trim()); } } return [...new Set(result)]; }
